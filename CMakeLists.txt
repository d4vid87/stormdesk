cmake_minimum_required(VERSION 3.21)
project(StormDesk VERSION 5.0.0 LANGUAGES CXX)
if(MSVC)
  add_compile_options(/utf-8)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(STORMDESK_GUI "Build the native Qt dashboard" ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Sql)
add_library(stormdesk_core cpp/core.cpp cpp/services.cpp cpp/http.cpp cpp/integrations.cpp cpp/advanced.cpp cpp/discovery.cpp)
file(GLOB_RECURSE STORMDESK_ASSETS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/site/*")
qt_add_resources(stormdesk_core siteassets PREFIX "/site" BASE "${CMAKE_CURRENT_SOURCE_DIR}/site" FILES ${STORMDESK_ASSETS})
target_include_directories(stormdesk_core PUBLIC cpp)
target_link_libraries(stormdesk_core PUBLIC Qt6::Core Qt6::Network Qt6::Sql)
target_compile_definitions(stormdesk_core PUBLIC STORMDESK_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(stormdesk_core PRIVATE /W4)
else()
  target_compile_options(stormdesk_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND AND NOT ANDROID)
  pkg_check_modules(MOSQUITTO QUIET IMPORTED_TARGET libmosquitto>=2.0)
  if(TARGET PkgConfig::MOSQUITTO)
    target_link_libraries(stormdesk_core PRIVATE PkgConfig::MOSQUITTO)
    target_compile_definitions(stormdesk_core PRIVATE STORMDESK_MQTT)
  endif()
endif()
if(NOT TARGET PkgConfig::MOSQUITTO)
  find_path(MOSQUITTO_INCLUDE_DIR mosquitto.h)
  find_library(MOSQUITTO_LIBRARY NAMES mosquitto libmosquitto)
  if(MOSQUITTO_INCLUDE_DIR AND MOSQUITTO_LIBRARY)
    target_include_directories(stormdesk_core PRIVATE "${MOSQUITTO_INCLUDE_DIR}")
    target_link_libraries(stormdesk_core PRIVATE "${MOSQUITTO_LIBRARY}")
    target_compile_definitions(stormdesk_core PRIVATE STORMDESK_MQTT)
  endif()
endif()
qt_add_executable(stormdesk cpp/main.cpp)
target_link_libraries(stormdesk PRIVATE stormdesk_core)
if(STORMDESK_GUI)
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  option(STORMDESK_WEBENGINE "Embed HookEcho radar using Qt WebEngine when available" ON)
  if(STORMDESK_WEBENGINE AND NOT ANDROID)
    find_package(Qt6 QUIET COMPONENTS WebEngineWidgets)
  endif()
  qt_add_resources(stormdesk_core nativefonts PREFIX "/fonts" BASE "${CMAKE_CURRENT_SOURCE_DIR}/cpp/fonts" FILES cpp/fonts/barlow-semi-condensed-500.ttf cpp/fonts/OFL.txt)
  target_sources(stormdesk PRIVATE cpp/window.cpp cpp/tools.cpp cpp/radar.cpp cpp/instruments.cpp cpp/setup.cpp)
  qt_add_resources(stormdesk appicons PREFIX "/" FILES app-icon.png)
  if(APPLE)
    set_target_properties(stormdesk PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.d4vid87.stormdesk.native")
  endif()
  target_link_libraries(stormdesk PRIVATE Qt6::Widgets)
  target_compile_definitions(stormdesk PRIVATE STORMDESK_GUI)
  if(TARGET Qt6::WebEngineWidgets)
    target_link_libraries(stormdesk PRIVATE Qt6::WebEngineWidgets)
    target_compile_definitions(stormdesk PRIVATE STORMDESK_WEBENGINE)
  else()
    message(STATUS "Qt WebEngine unavailable: HookEcho opens in the system browser")
  endif()
  find_package(Qt6 QUIET COMPONENTS TextToSpeech)
  if(TARGET Qt6::TextToSpeech)
    target_link_libraries(stormdesk PRIVATE Qt6::TextToSpeech)
    target_compile_definitions(stormdesk PRIVATE STORMDESK_SPEECH)
  endif()
endif()
if(ANDROID)
  set_target_properties(stormdesk PROPERTIES QT_ANDROID_PACKAGE_NAME "io.github.d4vid87.stormdesk.qt"
    QT_ANDROID_VERSION_NAME "${PROJECT_VERSION}" QT_ANDROID_VERSION_CODE "50001")
  set(STORMDESK_ANDROID_OPENSSL_DIR "" CACHE PATH "Path to the Qt-recommended KDAB android_openssl package")
  if(NOT EXISTS "${STORMDESK_ANDROID_OPENSSL_DIR}/android_openssl.cmake")
    message(FATAL_ERROR "Android HTTPS requires OpenSSL. Set STORMDESK_ANDROID_OPENSSL_DIR; see docs/native-cpp.md.")
  endif()
  include("${STORMDESK_ANDROID_OPENSSL_DIR}/android_openssl.cmake")
  add_android_openssl_libraries(stormdesk)
endif()
include(GNUInstallDirs)
install(TARGETS stormdesk RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} BUNDLE DESTINATION .)
install(FILES cpp/stormdesk.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES app-icon.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/512x512/apps RENAME stormdesk.png)
if(WIN32 AND STORMDESK_GUI)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
  include(InstallRequiredSystemLibraries)
  find_program(STORMDESK_WINDEPLOYQT windeployqt REQUIRED)
  install(CODE "execute_process(COMMAND \"${STORMDESK_WINDEPLOYQT}\" --no-translations \"\${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_BINDIR}/stormdesk.exe\" COMMAND_ERROR_IS_FATAL ANY)")
endif()
install(FILES LICENSE cpp/fonts/OFL.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/stormdesk)
set(CPACK_PACKAGE_NAME "StormDesk")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_CONTACT "https://github.com/d4vid87/stormdesk/issues")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native Qt weather dashboard with HookEcho radar")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGE_INSTALL_DIRECTORY "StormDesk")
if(WIN32)
  set(CPACK_GENERATOR "ZIP")
elseif(UNIX AND NOT APPLE AND NOT ANDROID)
  set(CPACK_GENERATOR "DEB")
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "libqt6sql6-sqlite")
endif()
include(CPack)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(stormdesk_tests cpp/tests.cpp)
  target_link_libraries(stormdesk_tests PRIVATE stormdesk_core Qt6::Test)
  add_test(NAME stormdesk_core COMMAND stormdesk_tests -nocrashhandler -o native-core-results.txt,txt)
  if(STORMDESK_GUI AND NOT ANDROID)
    add_executable(stormdesk_ui_tests cpp/ui_tests.cpp cpp/window.cpp cpp/tools.cpp cpp/radar.cpp cpp/instruments.cpp cpp/setup.cpp)
    target_link_libraries(stormdesk_ui_tests PRIVATE stormdesk_core Qt6::Widgets Qt6::Test)
    add_test(NAME stormdesk_ui COMMAND stormdesk_ui_tests -nocrashhandler -o native-ui-results.txt,txt)
    if(APPLE)
      set_tests_properties(stormdesk_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=cocoa")
    else()
      set_tests_properties(stormdesk_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
  endif()
endif()
